cmake_minimum_required(VERSION 3.22)
project(MettaVoxAD_Bundle VERSION 3.0.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 7.0.12 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
# JUCE module interface sources would otherwise be compiled again by every
# consumer. Keep module headers/definitions/system libraries, but materialise
# implementation sources exactly once in the shared static core.
set(MV_MODULES juce_audio_utils juce_dsp juce_audio_processors juce_audio_formats
    juce_audio_devices juce_audio_basics juce_gui_extra juce_gui_basics
    juce_graphics juce_events juce_data_structures juce_core)
set(MV_MODULE_SOURCES)
foreach(module IN LISTS MV_MODULES)
 get_target_property(module_sources ${module} INTERFACE_SOURCES)
 if(module_sources)
  list(APPEND MV_MODULE_SOURCES ${module_sources})
 endif()
 set_property(TARGET ${module} PROPERTY INTERFACE_SOURCES "")
endforeach()
add_library(mv3core STATIC Source/Processor.cpp Source/Editor.cpp ${MV_MODULE_SOURCES})
target_include_directories(mv3core PUBLIC Source)
target_compile_definitions(mv3core PUBLIC JUCE_PLUGINHOST_VST3=1
        JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(mv3core PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
if(MSVC)
 target_compile_options(mv3core PUBLIC /utf-8)
endif()
set(MV_NAMES DeCon SessionStrip TransientDynamix SonicMatrix SpaceWeaver Exciter808 MaximumCeiling)
set(MV_BRANDS Pure Vocal Impact Morph Orbit Velvet Apex)
set(MV_CODES Mv30 Mv31 Mv32 Mv33 Mv34 Mv35 Mv36)
set(kind 0)
foreach(name IN LISTS MV_NAMES)
 list(GET MV_CODES ${kind} code)
 list(GET MV_BRANDS ${kind} brand)
 juce_add_plugin(${name} COMPANY_NAME "MettaVoxAD Audio Engineering" BUNDLE_ID "com.mettavoxad.bundle.${name}" PLUGIN_MANUFACTURER_CODE Mtvx PLUGIN_CODE ${code} FORMATS VST3 PRODUCT_NAME "MettaVoxAD ${brand}" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx)
 if(TARGET juce_vst3_helper)
  target_link_libraries(juce_vst3_helper PRIVATE mv3core)
 endif()
 target_sources(${name} PRIVATE Source/Entry.cpp)
 target_link_libraries(${name} PRIVATE mv3core)
 target_include_directories(${name} PRIVATE $<TARGET_PROPERTY:mv3core,INCLUDE_DIRECTORIES>)
 target_compile_definitions(${name} PRIVATE MV_KIND=${kind} $<TARGET_PROPERTY:mv3core,COMPILE_DEFINITIONS>)
 if(WIN32)
  add_library(${name}_Legacy MODULE Source/Legacy.cpp Source/LegacyVst.def)
  target_link_libraries(${name}_Legacy PRIVATE mv3core)
  target_include_directories(${name}_Legacy PRIVATE $<TARGET_PROPERTY:mv3core,INCLUDE_DIRECTORIES>)
  target_compile_definitions(${name}_Legacy PRIVATE MV_KIND=${kind} $<TARGET_PROPERTY:mv3core,COMPILE_DEFINITIONS>)
  set_target_properties(${name}_Legacy PROPERTIES PREFIX "" OUTPUT_NAME "MettaVoxAD ${brand}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Legacy/$<CONFIG>" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Legacy/$<CONFIG>")
 endif()
 math(EXPR kind "${kind}+1")
endforeach()
add_executable(bundle_checks Tests/BundleChecks.cpp)
target_link_libraries(bundle_checks PRIVATE mv3core)
target_include_directories(bundle_checks PRIVATE $<TARGET_PROPERTY:mv3core,INCLUDE_DIRECTORIES>)
target_compile_definitions(bundle_checks PRIVATE $<TARGET_PROPERTY:mv3core,COMPILE_DEFINITIONS>)
if(WIN32)
 add_executable(bundle_legacy_checks Tests/LegacyChecks.cpp)
 target_include_directories(bundle_legacy_checks PRIVATE Source)
 target_link_libraries(bundle_legacy_checks PRIVATE user32 gdi32)
endif()

if(WIN32)
 add_executable(vst3_smoke EXCLUDE_FROM_ALL Tests/Vst3Smoke.cpp)
 target_link_libraries(vst3_smoke PRIVATE mv3core user32 gdi32)
 target_include_directories(vst3_smoke PRIVATE Source $<TARGET_PROPERTY:mv3core,INCLUDE_DIRECTORIES>)
 target_compile_definitions(vst3_smoke PRIVATE $<TARGET_PROPERTY:mv3core,COMPILE_DEFINITIONS>)
endif()
